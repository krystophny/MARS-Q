cmake_minimum_required(VERSION 3.20)
project(mars_profile_bracket LANGUAGES Fortran)
enable_testing()
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../CheaseMerge")
set(CHEASE_SOURCE "${NATIVE}/chease.f" CACHE FILEPATH "Native monolithic source")
file(READ "${CHEASE_SOURCE}" MONOLITHIC)
set(SELECTED "")
foreach(NAME PPSPLN RESETI)
  string(FIND "${MONOLITHIC}" "         SUBROUTINE ${NAME}(" BEGIN)
  string(SUBSTRING "${MONOLITHIC}" ${BEGIN} -1 TAIL)
  string(FIND "${TAIL}" "\nC*DECK" END)
  string(SUBSTRING "${TAIL}" 0 ${END} ROUTINE)
  string(APPEND SELECTED "${ROUTINE}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_profile.f" "${SELECTED}")
add_library(native_profile "${CMAKE_CURRENT_BINARY_DIR}/native_profile.f"
  baseline_ppspln.f scale_fixture.f)
target_include_directories(native_profile PUBLIC "${NATIVE}")
add_executable(profile_oracle oracle.F90)
target_compile_definitions(profile_oracle PRIVATE MARS)
target_link_libraries(profile_oracle PRIVATE native_profile)
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(native_profile PRIVATE -fdefault-real-8 -fdefault-double-8 -ffixed-line-length-none)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(native_profile PRIVATE -fcheck=all -ffpe-trap=invalid,zero,overflow)
    target_compile_options(profile_oracle PRIVATE -fcheck=all -ffpe-trap=invalid,zero,overflow)
  endif()
endif()
add_test(NAME native_profile_cubic_derivative_parent_bytes COMMAND profile_oracle)
