cmake_minimum_required(VERSION 3.20)
project(mars_chease_smoothing_permutation LANGUAGES Fortran)
enable_testing()
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../CheaseMerge")
set(CHEASE_SOURCE "${NATIVE}/chease.f" CACHE FILEPATH "Native monolithic source")
file(READ "${CHEASE_SOURCE}" MONOLITHIC)
set(SELECTED "")
foreach(NAME SMOOTH MSPLINE MSPLCY TRIDAGM TRICYCM)
  string(FIND "${MONOLITHIC}" "         SUBROUTINE ${NAME}" BEGIN)
  if(BEGIN LESS 0)
    message(FATAL_ERROR "Missing native ${NAME}")
  endif()
  string(SUBSTRING "${MONOLITHIC}" ${BEGIN} -1 TAIL)
  string(FIND "${TAIL}" "
C*DECK" END)
  string(SUBSTRING "${TAIL}" 0 ${END} ROUTINE)
  string(APPEND SELECTED "${ROUTINE}
")
endforeach()
string(FIND "${MONOLITHIC}" "         FUNCTION ISAMIN" BEGIN)
string(SUBSTRING "${MONOLITHIC}" ${BEGIN} -1 TAIL)
string(FIND "${TAIL}" "
C*DECK" END)
string(SUBSTRING "${TAIL}" 0 ${END} ROUTINE)
string(APPEND SELECTED "${ROUTINE}
")
string(REPLACE "SUBROUTINE SMOOTH" "SUBROUTINE NATIVE_SMOOTH" SELECTED "${SELECTED}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_smooth.f" "${SELECTED}")
configure_file("${NATIVE}/COMDIM.inc" "${CMAKE_CURRENT_BINARY_DIR}/COMDIM.inc" COPYONLY)
configure_file("${NATIVE}/DECLAR.inc" "${CMAKE_CURRENT_BINARY_DIR}/DECLAR.inc" COPYONLY)
foreach(NAME COMCON COMINT COMPHY)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.inc" "C Unused in selected native routines
")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/COMBLA.inc" "         COMMON /FIXC/ CPSI(NP4NST),CPSICL(NP4NST)
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/COMESH.inc" "         COMMON /FIXM/ CSIG(NSP1),CT(NTP1)
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/COMNUM.inc" "         COMMON /FIXN/ NS,NT,NS1,NT1
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/COMSOL.inc" "         COMMON /FIXP/ NUPDWN(NSNT)
")
add_executable(smoothing_oracle globals_fixture.f90 oracle.f90 bridge.f90 wrapper.f
  "${CMAKE_CURRENT_BINARY_DIR}/native_smooth.f")
target_include_directories(smoothing_oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(smoothing_oracle PRIVATE -fdefault-real-8 -fdefault-double-8
    -ffixed-line-length-none -std=legacy -fcheck=all -ffpe-trap=invalid,zero,overflow)
endif()
add_test(NAME smoothing_all_four_jet_components COMMAND smoothing_oracle)
