cmake_minimum_required(VERSION 3.20)
project(mars_chease_axis_field LANGUAGES Fortran)
enable_testing()
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../CheaseMerge")
set(CHEASE_SOURCE "${NATIVE}/chease.f" CACHE FILEPATH "Native MAPPIN source")
file(READ "${CHEASE_SOURCE}" MONOLITHIC)
string(FIND "${MONOLITHIC}" "T0    = FCCCC0(TMF(1)" BEGIN)
if(BEGIN LESS 0)
  message(FATAL_ERROR "Missing native MAPPIN K1 axis-field assignment")
endif()
string(SUBSTRING "${MONOLITHIC}" ${BEGIN} -1 TAIL)
string(FIND "${TAIL}" "\n" FIRST_END)
math(EXPR SECOND_BEGIN "${FIRST_END}+1")
string(SUBSTRING "${TAIL}" ${SECOND_BEGIN} -1 SECOND)
string(FIND "${SECOND}" "\n" SECOND_END)
math(EXPR LENGTH "${SECOND_BEGIN}+${SECOND_END}")
string(SUBSTRING "${TAIL}" 0 ${LENGTH} ASSIGNMENT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_axis_field.f"
"      DOUBLE PRECISION FUNCTION NATIVE_AXIS_FIELD(TMF,CSM)\n      IMPLICIT DOUBLE PRECISION (A-H,O-Z)\n      DIMENSION TMF(4),CSM(4)\n      PARAMETER (RC0P=0D0)\n      INCLUDE 'CUCCCC.inc'\n         ${ASSIGNMENT}\n      NATIVE_AXIS_FIELD=T0\n      END\n")
configure_file("${NATIVE}/CUCCCC.inc" "${CMAKE_CURRENT_BINARY_DIR}/CUCCCC.inc" COPYONLY)
add_executable(axis_field_oracle oracle.f90 "${CMAKE_CURRENT_BINARY_DIR}/native_axis_field.f")
target_include_directories(axis_field_oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(axis_field_oracle PRIVATE -ffixed-line-length-none -fcheck=all -ffpe-trap=invalid,zero,overflow)
endif()
add_test(NAME native_mappin_axis_field_polynomials COMMAND axis_field_oracle)
