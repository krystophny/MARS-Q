cmake_minimum_required(VERSION 3.20)
project(mars_chease_contiguous_ldlt LANGUAGES Fortran)
enable_testing()
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../CheaseMerge")
set(CHEASE_SOURCE "${NATIVE}/chease.f" CACHE FILEPATH "Native monolithic source")
file(READ "${CHEASE_SOURCE}" MONOLITHIC)
set(SELECTED "")
foreach(NAME ALDLT LYV DWY LTXW SAXPY SDOT)
  if(NAME STREQUAL "SDOT")
    set(DECLARATION "         FUNCTION ${NAME}")
  else()
    set(DECLARATION "         SUBROUTINE ${NAME}")
  endif()
  string(FIND "${MONOLITHIC}" "${DECLARATION}" BEGIN)
  if(BEGIN LESS 0)
    message(FATAL_ERROR "Missing native ${NAME}")
  endif()
  string(SUBSTRING "${MONOLITHIC}" ${BEGIN} -1 TAIL)
  string(FIND "${TAIL}" "\nC*DECK" END)
  string(SUBSTRING "${TAIL}" 0 ${END} ROUTINE)
  string(APPEND SELECTED "${ROUTINE}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_band.f" "${SELECTED}")
configure_file("${NATIVE}/DECLAR.inc" "${CMAKE_CURRENT_BINARY_DIR}/DECLAR.inc" COPYONLY)
add_library(native_band precision_fixture.f90 baseline_aldlt.f
  "${CMAKE_CURRENT_BINARY_DIR}/native_band.f")
set_target_properties(native_band PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(native_band PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(band_oracle oracle.f90)
target_link_libraries(band_oracle PRIVATE native_band)
add_executable(band_timing timing.f90)
target_link_libraries(band_timing PRIVATE native_band)
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  foreach(TARGET native_band band_oracle band_timing)
    target_compile_options(${TARGET} PRIVATE -fdefault-real-8 -fdefault-double-8
      -ffixed-line-length-none -std=legacy)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
      target_compile_options(${TARGET} PRIVATE -fcheck=all -ffpe-trap=invalid,zero,overflow)
    endif()
  endforeach()
endif()
add_test(NAME band_dense_independent_and_bitwise COMMAND band_oracle)
add_test(NAME band_empty_singleton_singular COMMAND band_oracle edge)
